052:     #now we define the basic roman numeral conversion functions, which are used by the main funcs

054:     # Basic Roman convert funcs (1-3999)
055:     def _basic_roman_to_int(self, roman_str):
056:         total = 0
057:         n = len(roman_str)
058:         for i in range(n): #for loop to iterate through the string
059:             ch = roman_str[i]
060:             if ch not in self.roman_to_int_dict:
061:                 raise ValueError("Invalid symbol '{}' in Roman numeral.".format(ch)) #we use a valueerror to catch bad people (who use bad symb)
062:             
063:             val = self.roman_to_int_dict[ch]
064:             # first rule of roman, the subtractive notation
065:             if (i + 1 < n) and (val < self.roman_to_int_dict.get(roman_str[i + 1], 0)):
066:                 total -= val
067:             else:
068:                 total += val
069:         return total
070: 
071:     def _basic_int_to_roman(self, num):
072:         #Converts a basic int into its numeral
073:         result = ""
074:         for val, symbol in self.int_to_roman_dict.items(): #we use the items() method to get the key-value pairs from the dictionary
075:             count = num // val #items allows us to iterate through the dictionary and get the key-value pairs
076:             result += symbol * count
077:             num = num % val #we then use mod to get the remain and continue
078:         return result

080:     def number_to_roman(self, number):
081:         #now convert any float / int into a full numeral str
082:         if abs(number) < 1e-9: #we have a rather large upper limit
083:             return "N" #this gives us us the zero numeral 
084: 
085:         sign = "-" if number < 0 else ""
086:         number = abs(number) #we then take the absolute value of the number to make it positive for conv
087: 
088:         whole = int(math.floor(number + 1e-9))
089:         frac = number - whole #then we get the fractional part of the number
090: 
091:         # Convert fractional part to nearest twelfth (uncia)
092:         twelfths = int(round(frac * 12))
093:         if twelfths == 12:
094:             whole += 1
095:             twelfths = 0
096: 
097:         if whole > 3999999: #we account for overflow, as the vinculum system only allows for 3,999,999 as the max
098:             raise OverflowError("Result exceeds maximum Vinculum limit (3,999,999).") #spell check did not know what this word meant
099: 
100:         # Build whole number part
101:         roman_whole = ""
102:         if whole >= 4000:
103:             thousands = whole // 1000 #here we use integer division to get the number of thousands or hundereds of thousands
104:             remainder = whole % 1000
105:             roman_whole = "({})".format(self._basic_int_to_roman(thousands)) #now we use format to build the vinculum part of the numeral
106:             if remainder > 0:
107:                 roman_whole += self._basic_int_to_roman(remainder) #then for any remainder we just use the basic conversion function
108:         elif whole > 0:
109:             roman_whole = self._basic_int_to_roman(whole) #then we call §the basic conversion function for any number less than 4000
110: 
111:         # Build fractional part
112:         roman_frac = self.twelfths_to_roman_dict[twelfths]
113: 
114:         #here we return the final numeral, with the sign, whole part, and fractional part
115:         if roman_whole == "" and roman_frac == "":
116:             return "N"
117:         return sign + roman_whole + roman_frac

119:     #new function to convert roman numeral to number, with validation (yipeeeee)

121:     def roman_to_number(self, roman_str):
122:         """
123:         Parses and strictly validates a Roman numeral string (including negatives,
124:         Vinculum brackets, and base-12 fractions) into a Python float/int.
125:         """
126:         raw_input = roman_str.strip().upper() #we have to strip out the whitespace and make it uppercase to make it easier to parse
127:         if not raw_input:
128:             raise ValueError("Empty Roman numeral input.") #then we check for empty input and raise a valueerror if it is empty
129: 
130:         if raw_input == "N": #we can now handle the zero numeral
131:             return 0
132: 
133:         # Check for negative sign
134:         sign = 1
135:         s = raw_input
136:         if s.startswith("-"):
137:             sign = -1
138:             s = s[1:].strip()
139: 
140:         # Extract any trailing base-12 fraction characters ('S', ':', '.') you can read more about this at [cite]
141:         frac_chars = ""
142:         while len(s) > 0 and s[-1] in ('S', ':', '.'): #pretty standard while loop to extract the fraction characters from the end of the string
143:             frac_chars = s[-1] + frac_chars
144:             s = s[:-1]
145: 
146:         twelfths = 0 #then we convert the fraction characters into a number of twelfths
147:         for ch in frac_chars:
148:             if ch == 'S':
149:                 twelfths += 6
150:             elif ch == ':':
151:                 twelfths += 2
152:             elif ch == '.':
153:                 twelfths += 1
154:         if twelfths >= 12:
155:             raise ValueError("Invalid fraction '{}' (>= 12/12).".format(frac_chars)) #just in case we have one more valueerror
156: 
157:         # Parse whole number part
158:         whole_val = 0
159:         if len(s) > 0:
160:             if s.startswith("("):
161:                 # Handle Vinculum brackets (thousands separator)
162:                 if ")" not in s:
163:                     raise ValueError("Unclosed Vinculum bracket '('.")
164:                 close_idx = s.index(")")
165:                 thousands_part = s[1:close_idx]
166:                 remainder_part = s[close_idx + 1:]
167:                 if not thousands_part: #in cases where the thousands is empty 
168:                     raise ValueError("Empty Vinculum brackets '()'.")
169:                 whole_val = self._basic_roman_to_int(thousands_part) * 1000 #then we multiply the thousands part by 1000 to get the whole value
170:                 if remainder_part:
171:                     whole_val += self._basic_roman_to_int(remainder_part)
172:             else:
173:                 whole_val = self._basic_roman_to_int(s) #or we can just use the basic conversion function if there are no vinculum brackets
174: 
175:         total = sign * (whole_val + (twelfths / 12.0)) #take the sign into account and add the whole value and the fractional value together to get the total
176: 
177:         # Canonical Round-Trip Validation!
178:         canonical = self.number_to_roman(total)
179:         if canonical != raw_input:
180:             raise ValueError("not valid roman notation")
181: 
182:         return int(total) if twelfths == 0 else total #return the total as an int if there are no twelfths, otherwise return it as a float

184:     #new function; this time we are going to evaluate the expression, which is the main function of the calculator
185:     #this is the main function that takes in the two roman numerals and the operator and returns the result in both roman and decimal form
186:     #it also handles the square root operator, which is a unary operator, and the other operators are binary operators
187: 
188:     def evaluate(self, num1_str, operator, num2_str=None):
189:         # we beign by executing an operation (+, -, *, /, ^, SQRT) and returns (roman_res, dec_res).
190:         val1 = self.roman_to_number(num1_str)
191: 
192:         if operator == "SQRT": #we handle the square root operator, which is a unary operator
193:             if val1 < 0:
194:                 raise ValueError("Cannot take square root of a negative numeral.") #theres going to be a lot of valueerrors in this func, but they are all necessary to catch the user when they do something dumb, which is likely
195:             res = math.sqrt(val1)
196:             return self.number_to_roman(res), res
197: 
198:         val2 = self.roman_to_number(num2_str) #we then convert the second numeral to a number, which is necessary for the binary operators
199: 
200:         #and now we sit and do the actual math, which is the main part of the function, and we use a series of if statements to determine which operator to use
201: 
202:         if operator == '+': #i wont comment every single one of these, as they are all pretty self explanatory
203:             res = val1 + val2
204:         elif operator == '-':
205:             res = val1 - val2
206:         elif operator == '*':
207:             res = val1 * val2
208:         elif operator == '/':
209:             if abs(val2) < 1e-9:
210:                 raise ZeroDivisionError("Division by N (Zero) is undefined.") #this one is important, as we dont want to divide by zero, i am told this is bad
211:             res = val1 / val2
212:         elif operator == '^':
213:             res = val1 ** val2
214:         else:
215:             raise ValueError("Unknown operator '{}'".format(operator)) #i mean, yeah just in case they try and idk like use grad
216: 
217:         return self.number_to_roman(res), res #and finally we spit out the result in both roman and decimal form

360:     def on_button_click(self, label):
361:         # Extract symbol from labels like "S (1/2)" -> "S"
362:         symbol = label.split(" ")[0]
363: 
364:         #now for the button click handler
365:         if symbol == "CLR": #we do this for clear
366:             self.first_operand = ""
367:             self.current_operator = ""
368:             self.current_input = ""
369:             self.just_evaluated = False
370:             self._update_displays()
371:             return
372: 
373:         if symbol == "DEL": #delete is a bit more complicated, as we have to check if we just evaluated an expression, in which case we clear the input, otherwise we just delete the last character
374:             if self.just_evaluated:
375:                 self.current_input = ""
376:                 self.just_evaluated = False
377:             else:
378:                 self.current_input = self.current_input[:-1]
379:             self._update_displays()
380:             return
381: 
382:         if symbol == "+/-": #then for the negative toggle, we just check if the current input starts with a negative sign, and if it does we remove it, otherwise we add it to the front
383:             if self.current_input.startswith("-"):
384:                 self.current_input = self.current_input[1:]
385:             elif self.current_input and self.current_input != "N":
386:                 self.current_input = "-" + self.current_input
387:             self._update_displays()
388:             return
389: 
390:         # Square Root operator
391:         if symbol == "SQRT":
392:             target = self.current_input if self.current_input else "N"
393:             try:
394:                 rom_res, dec_res = self.engine.evaluate(target, "SQRT") #we take the current input and evaluate it with the square root operator, which is a unary operator
395:                 log_entry = "SQRT({}) = {}  [{:g}]".format(target, rom_res, dec_res)
396:                 self.history_list.insert(0, log_entry)
397:                 self.current_input = rom_res
398:                 self.first_operand = ""
399:                 self.current_operator = ""
400:                 self.just_evaluated = True #we set the just_evaluated flag to true so that the next time the user enters a numeral, it will clear the current input
401:                 self._update_displays()
402:             except Exception as e:
403:                 messagebox.showerror("Calculation Error", str(e)) #finally we catch any errors and show them in a messagebox
404:             return
405: 
406:         # Binary operators (+, -, *, /, ^)
407:         if symbol in ('+', '-', '*', '/', '^'):
408:             if not self.current_input:
409:                 self.current_input = "N"
410:             try:
411:                 # Validate first operand before locking it in
412:                 self.engine.roman_to_number(self.current_input)
413:                 self.first_operand = self.current_input
414:                 self.current_operator = symbol
415:                 self.current_input = ""
416:                 self.just_evaluated = False
417:                 self._update_displays()
418:             except Exception as e:
419:                 messagebox.showwarning("Invalid Roman Numeral", str(e)) #same thing here, we catch any errors and show them in a messagebox
420:             return
421:

423:         if symbol == "=":
424:             if not self.first_operand or not self.current_operator: #now we check if we have a first operand and an operator, if not we just return

427:             try:
428:                 rom_res, dec_res = self.engine.evaluate(
429:                     self.first_operand, self.current_operator, second_operand
430:                 ) #we then evaluate the expression with the first operand, operator, and second operand, and get the result in both roman and decimal form
431:                 log_entry = "{} {} {} = {}  [{:g}]".format(
432:                     self.first_operand, self.current_operator, second_operand, rom_res, dec_res

441:                 self.just_evaluated = True
